# ---- compiled by goto.sh: 2 label(s), 3 segment(s) ----
declare -g __GOTO_SHELL=$BASHPID __GOTO_PC=__gt_entry
declare -ga __GOTO_STACK=()
declare -g __GOTO_RC=0
__gt_rc() { return "${1:-0}"; }
declare -F __gt_fault > /dev/null || __gt_fault() {
	printf 'goto: fatal: `goto %s` executed in a subshell or pipeline (pid %s != %s)\n' "$1" "$BASHPID" "$__GOTO_SHELL" >&2
	kill -0 "$__GOTO_SHELL" 2>/dev/null &&
	    kill -s TERM "$__GOTO_SHELL" 2>/dev/null
	exit 70
}
while :; do
  case $__GOTO_PC in
    (__gt_entry)
      __GOTO_PC=skip
      (( __GOTO_RC == 0 )) || { [[ $- == *e* ]] || __gt_rc "$__GOTO_RC"; }
    source "$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)/../goto.sh";
    echo line 1;
    { __GOTO_RC=$?; __GOTO_PC=skip; [[ $BASHPID == "$__GOTO_SHELL" ]] || __gt_fault skip; continue 1; };
    echo "you should never see this";
      __GOTO_RC=$?
      ;;
    (skip)
      __GOTO_PC=top
      (( __GOTO_RC == 0 )) || { [[ $- == *e* ]] || __gt_rc "$__GOTO_RC"; }
    echo line 3;
    echo "--- backwards: a loop built out of nothing but goto ---";
    i=0;
    state="untouched by any re-exec";
      __GOTO_RC=$?
      ;;
    (top)
      __GOTO_PC=__gt_END
      (( __GOTO_RC == 0 )) || { [[ $- == *e* ]] || __gt_rc "$__GOTO_RC"; }
    echo "iteration $i  ($state)";
    : $(( i++ ));
    (( i < 3 )) && { __GOTO_RC=$?; __GOTO_PC=top; [[ $BASHPID == "$__GOTO_SHELL" ]] || __gt_fault top; continue 1; };
    echo "done, and \$state survived: $state"
      __GOTO_RC=$?
      ;;
    (__gt_END) break ;;
    (*) printf "goto: no such label: %s\n" "$__GOTO_PC" >&2; exit 2 ;;
  esac
done
(exit "$__GOTO_RC")
