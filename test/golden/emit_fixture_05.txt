	kill -0 "$__GOTO_SHELL" 2>/dev/null &&
	    kill -s TERM "$__GOTO_SHELL" 2>/dev/null

}

    (*) printf "goto: no such label: %s\n" "$__GOTO_PC" >&2; exit 2 ;;
